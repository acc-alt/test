如何建立您專屬的 Google Apps Script 雲端資料庫

本系統完全運行在您的瀏覽器，無任何本地端資料庫。資料將即時、安全地儲存在您自己的 Google 試算表中。以下是快速建置教學，完全隱密且免費：

步驟 1：建立 Google 試算表 建立一份全新空白的 Google 試算表，並在底部建立四個工作表（分頁），分別命名為：
land sea fleet
步驟 2：開啟 Google Apps Script (GAS) 編輯器 在該試算表上方選單，點選 擴充功能 (Extensions) > Apps Script。
步驟 3：貼入以下後端程式碼 請清空 Apps Script 編輯器內原本的所有內容，將下方代碼完全複製並貼上： 
/* Google Apps Script 後端程式碼 */
function doGet(e) {
  var action = e.parameter.action;
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action === "read") {
    var data = {};
    var sheets = ["land", "sea", "fleet"];
    sheets.forEach(function(s) {
      var sh = sheet.getSheetByName(s);
      if (sh) {
        var range = sh.getDataRange();
        if (range.getLastRow() > 1) {
          var values = range.getValues();
          var headers = values[0];
          data[s] = values.slice(1).map(function(row) {
            var obj = {};
            headers.forEach(function(header, idx) {
              obj[header] = row[idx];
            });
            return obj;
          });
        } else {
          data[s] = [];
        }
      } else {
        data[s] = [];
      }
    });
    return ContentService.createTextOutput(JSON.stringify({status: "success", data: data}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  try {
    var params = JSON.parse(e.postData.contents);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: "格式錯誤"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var action = params.action;
  var targetSheet = params.sheet;
  
  var sh = sheet.getSheetByName(targetSheet);
  if (!sh) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: "找不到指定工作表"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === "create") {
    var rowData = params.data;
    
    // 【修正】：將時間格式化為台灣時區 (GMT+8)，並且僅顯示年月日
    rowData.timestamp = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd");
    
    var headers = [];
    
    // 【修正】：若工作表為完全空白，getLastColumn() 為 0，直接取 Range 會報錯
    if (sh.getLastColumn() > 0) {
      headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
    }
    
    // 若工作表全新無欄位，自動從第一筆資料初始化表頭
    if (headers.length === 0) {
      headers = Object.keys(rowData);
      sh.appendRow(headers);
    }
    
    var newRow = headers.map(function(header) {
      return rowData[header] !== undefined ? rowData[header] : "";
    });
    
    sh.appendRow(newRow);
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  if (action === "delete") {
    var idToDelete = params.id;
    // 【修正】：檢查工作表是否為空
    if (sh.getLastRow() === 0) {
      return ContentService.createTextOutput(JSON.stringify({status: "success"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var values = sh.getDataRange().getValues();
    var headers = values[0];
    var idColIdx = headers.indexOf("id");
    
    if (idColIdx === -1) {
      return ContentService.createTextOutput(JSON.stringify({status: "error", message: "找不到識別ID欄位"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    for (var i = 1; i < values.length; i++) {
      if (values[i][idColIdx].toString() === idToDelete.toString()) {
        sh.deleteRow(i + 1); // Sheets indexed from 1
        return ContentService.createTextOutput(JSON.stringify({status: "success"}))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: "找不到欲刪除的ID"}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
步驟 4：儲存並部署為 Web App 1. 點擊 Apps Script 視窗上方的「儲存 (Save)」磁碟圖案。
2. 點擊右上角藍色 部署 (Deploy) > 新增部署 (New deployment)。
3. 齒輪處選擇「網頁應用程式 (Web app)」。
4. 描述填寫：`Carbon Database`。
5. 「將本網頁應用程式部署為」選擇：您的帳戶 (您的 Gmail)。
6. 「誰有權存取本網頁應用程式」務必選擇：所有人 (Anyone)。 (這非常重要，才能允許本 HTML 系統寫入數據)。
7. 點擊「部署」。首次部署需要「授予授權」，請在彈出的 Google 視窗選擇「進階」並允許存取該試算表。
8. 部署成功後，複製「網頁應用程式網址 (Web app URL)」。
步驟 5：貼入此系統進行安全連結 點擊網頁最上方右上角的 齒輪按鈕，將複製的網址貼入文字方塊中並點擊儲存。 系統會立即載入並連線，此網址完全儲存於您本地瀏覽器快取，他人無從得知，安全且隱私！
